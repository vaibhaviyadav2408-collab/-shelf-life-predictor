@app.route('/login', methods=['GET', 'POST'])
def login():

        return redirect(url_for('dashboard'))
    return render_template('login.html')

@app.route('/register', methods=['GET', 'POST'])
def register():
    if request.method == 'POST':

@app.route('/dashboard')
def dashboard():
    total = len(sample_products)
    fresh = sum(1 for p in sample_products if p['status'] == 'Fresh')
    expiring = sum(1 for p in sample_products if p['status'] == 'Expiring Soon')
    expired = sum(1 for p in sample_products if p['status'] == 'Expired')
    return render_template('dashboard.html', total=total, fresh=fresh, expiring=expiring, expired=expired, products=sample_products)
